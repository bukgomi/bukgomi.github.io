# 폴더명이 날짜면 6자리로 통일 (2025-03-04 · 25.03.04 → 250304)
function 날짜6 ($s) {
  if ($s -match &#x27;^(?:19|20|21)?(\d{2})[-._]?(0[1-9]|1[0-2])[-._]?(…)$') {
    "$($Matches[1])$($Matches[2])$($Matches[3])" } else { $s }
}

# 하위 폴더 경로를 접두어로: 2025-03-04\학교A정산\영수증.jpg
#   → 250304_학교A정산_영수증.jpg
$조각 = $_.DirectoryName.Substring($루트.Length) -split &#x27;[\\/]+' |
        Where-Object { $_ } | ForEach-Object { 날짜6 $_ }
$접두 = $조각 -join &#x27;_'

# 이미 붙은 접두어는 벗겨냄 — 여러 번 실행해도 중복 안 붙음
while ($접두 -and $끝.StartsWith("${접두}_")) {
  $끝 = $끝.Substring($접두.Length + 1)
}

# 같은 이름이 이미 있으면 덮어쓰지 않고 빨간색 경고

재구성 예시 — 코드 보기 (실사용 로직 그대로) 
const 폴더ID  = '여기에_폴더ID';
const 발신자  = ['billing@거래처A.co.kr', '@거래처B.com'];  // '@도메인'은 회사 전체
const 시작일  = '2026-08-01';   // 이 날 포함
const 종료일  = '2026-08-31';   // 이 날 포함

function 첨부저장() {
  const folder = DriveApp.getFolderById(폴더ID);
  const from = new Date(시작일 + 'T00:00:00+09:00');
  const to   = new Date(종료일 + 'T23:59:59+09:00');
  const q = 'has:attachment {' + 발신자.map(s => 'from:' + s).join(' ') + '}'
          + ' after:'  + ymd(new Date(from - 86400000))
          + ' before:' + ymd(new Date(+to + 86400000));
  GmailApp.search(q, 0, 50).forEach(thread => {
    thread.getMessages().forEach(msg => {
      const d = msg.getDate();
      if (d < from || d > to) return;              // 날짜 재확인
      if (!발신자맞음(msg.getFrom())) return;       // 발신자 재확인
      msg.getAttachments({ includeInlineImages: false }).forEach(att => {
        const name = ymd2(d) + '_' + att.getName();
        if (folder.getFilesByName(name).hasNext()) return;   // 중복 방지
        folder.createFile(att.copyBlob().setName(name));
        Logger.log('저장 ' + name);
      });
    });
  });
}

function 발신자맞음(f) {
  const m = (f.match(/<(.+?)>/) || [0, f])[1].toLowerCase().trim();
  return 발신자.some(s => {
    s = s.toLowerCase();
    return s[0] === '@' ? m.slice(-s.length) === s : m === s;
  });
}

function ymd(d)  { return Utilities.formatDate(d, 'Asia/Seoul', 'yyyy/MM/dd'); }
function ymd2(d) { return Utilities.formatDate(d, 'Asia/Seoul', 'yyyyMMdd'); }

PROBLEM부모님의 장부에서 시작된 프로젝트

퇴사 후, 사업을 하시는 부모님이 제가 만들어드린 엑셀 장부를 여전히 한 칸씩 수기로 채우고 계셨습니다. 작은 회사에 상용 ERP는 비용도, 복잡함도 부담이라는 것을 그때 체감했습니다. 무료로, 설치 부담 없이 쓸 수 있는 한국어 프로그램이 필요하다고 판단했습니다.

SOLUTION13개 모듈, 배포 2종, 3중 데이터 안전장치

거래처·재고, 매출·매입, 수금·지급(미수금·거래처 원장), 거래명세표, 회계 리포트, 장부·자체기장(연 마감) 등 13개 모듈을 업종에 맞게 켜고 끌 수 있습니다. 서버 없이 단일 HTML 파일과 Windows 설치형(.exe) 두 가지로 배포하며, 자동 저장·일별 백업·구글드라이브 동기화로 데이터를 보호합니다.

미니 ERP — 설계 상세

만들지 않을 것부터 정했습니다

세금 신고, 전자세금계산서 국세청 연동, 4대보험 계산은 의도적으로 제외했습니다. 매년 바뀌는 법령과 요율을 개인이 책임질 수 없고, 그 부분은 세무 전문가의 영역이기 때문입니다. 범위를 지키는 것도 설계라고 생각합니다.

핵심 기능 3가지 — 기존 ERP의 불편을 이렇게 바꿨습니다

핵심 기능 1

설치·서버·계정 없이, 파일 하나로 시작

기존 ERP의 불편설치와 서버 구축, 계정 발급, 도입 교육까지 거쳐야 시작할 수 있습니다. IT 담당자가 없는 작은 회사에는 비용보다 이 도입 과정 자체가 더 큰 장벽입니다.
미니 ERP의 해결단일 HTML 파일을 내려받아 더블클릭하면 끝입니다. 서버도 계정도 없고, 데이터는 내 PC 폴더에 저장되며 구글드라이브가 알아서 동기화·백업합니다. 설치형(.exe)도 같은 데이터로 호환됩니다.
차별점 — 도입 비용(무료)만이 아니라 도입 난이도를 없앴습니다. 컴퓨터 비전문가 기준으로 설계해, 매뉴얼도 그림처럼 따라 하는 A4 인쇄용으로 만들었습니다.
핵심 기능 2

경리 실무의 언어로 만든 수금·미수금 관리

기존 ERP의 불편회계 전문용어와 복잡한 전표 구조가 기본값입니다. 정작 작은 회사의 실제 고민인 부분 수금, 거래처별 잔금, 미수금 관리는 여러 단계 메뉴 속에 숨어 있습니다.
미니 ERP의 해결부분 수금과 잔금, 미수금 연령 분석, 거래처 원장, 거래처 단위 일괄 수금(FIFO 자동 배분)을 첫 화면 가까이에 배치했습니다. 수금 업로드 양식도 통장 기록 용어(입금액) 그대로입니다.
차별점 — 거래명세표를 실물 양식 그대로(A4 가로 2부, 전잔금·입금·총미수잔액) 인쇄합니다. 20억 정산 실무에서 매일 쓰던 사람이 만들었기 때문에 가능한 디테일입니다.
핵심 기능 3

데이터의 주인이 사용자인 구조

기존 ERP의 불편데이터가 업체 서버에 있어 해지하면 접근이 어렵고, 백업이 어떻게 되는지 사용자는 알 수 없습니다. 잘못 입력한 자료의 복구도 업체에 요청해야 합니다.
미니 ERP의 해결모든 데이터가 사람이 읽을 수 있는 JSON 파일로 내 폴더에 저장됩니다. 자동 저장, 일별 백업 30일 보관, 구글드라이브 동기화의 3중 안전장치가 있고, 백업 파일로 언제든 되돌릴 수 있습니다.
차별점 — 세무사 전달 패키지(증빙·장부 묶음)를 자동 생성하고, 연도 마감·이월과 장부 아카이브까지 지원합니다. 오픈소스(MIT)라 업체 종속 자체가 없습니다.

공정한 비교를 위해: 세금 신고·국세청 연동·다중 사용자 동시 편집은 상용 ERP가 강하고, 미니 ERP는 의도적으로 범위에서 제외했습니다. 미니 ERP의 자리는 "엑셀로는 부족하지만 상용 ERP는 부담스러운" 5~30인 회사입니다.
